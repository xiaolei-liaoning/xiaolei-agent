复杂任务 · JS脚本编排流程
agent()/parallel()/pipeline()

// workflow.js — 多Agent工作流示例
export const workflow = async (ctx) => {
  // 第一步：并行搜索和读取
  const [hot, data] = await ctx.parallel(
    () => ctx.agent("web_search", { query: "热搜" }),
    () => ctx.agent("read_file", { path: "data.json" })
  );
  // 第二步：分析聚合
  const report = await ctx.agent("分析", { hot, data });
  // 第三步：生成输出
  return await ctx.agent("writer", { report });
};
